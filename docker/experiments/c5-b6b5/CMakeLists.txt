cmake_minimum_required(VERSION 3.16)
project(c5_b6b5 LANGUAGES CXX)
find_package(ignition-gazebo6 6.18.0 EXACT REQUIRED)
add_library(c5-actuator-diff-drive SHARED DiffDrive.cc)
target_link_libraries(c5-actuator-diff-drive PRIVATE ignition-gazebo6::ignition-gazebo6)
target_compile_features(c5-actuator-diff-drive PRIVATE cxx_std_17)
target_compile_options(c5-actuator-diff-drive PRIVATE -ffile-prefix-map=${CMAKE_SOURCE_DIR}=/src)
add_executable(lease-test test_lease.cc)
target_compile_features(lease-test PRIVATE cxx_std_17)
enable_testing()
add_test(NAME zero-only-lease COMMAND lease-test)
add_executable(native-test test_native.cc)
target_link_libraries(native-test PRIVATE ignition-gazebo6::ignition-gazebo6)
target_compile_features(native-test PRIVATE cxx_std_17)
add_test(NAME native-zero-target COMMAND native-test)
find_package(ignition-plugin1 REQUIRED COMPONENTS loader)
add_executable(load-plugin load_plugin.cc)
target_link_libraries(load-plugin PRIVATE ignition-gazebo6::ignition-gazebo6 ignition-plugin1::loader)
target_compile_features(load-plugin PRIVATE cxx_std_17)
