cmake_minimum_required(VERSION 3.16)
project(c5_b6b7 LANGUAGES CXX)
find_package(ignition-gazebo6 6.18.0 EXACT REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(c5-authorized-zero-drive SHARED ../c5-b6b5/DiffDrive.cc)
target_compile_definitions(c5-authorized-zero-drive PRIVATE C5_AUTH_ZERO_ONLY=1
  DiffDrive=C5AuthorizedZeroDiffDrive DiffDrivePrivate=C5AuthorizedZeroDiffDrivePrivate)
target_link_libraries(c5-authorized-zero-drive PRIVATE ignition-gazebo6::ignition-gazebo6 OpenSSL::Crypto nlohmann_json::nlohmann_json)
add_executable(authorization-test test_authorization.cc)
target_link_libraries(authorization-test PRIVATE OpenSSL::Crypto nlohmann_json::nlohmann_json)
add_executable(native-test test_native.cc)
target_compile_definitions(native-test PRIVATE C5_AUTH_ZERO_ONLY=1
  DiffDrive=C5AuthorizedZeroDiffDrive DiffDrivePrivate=C5AuthorizedZeroDiffDrivePrivate)
target_link_libraries(native-test PRIVATE ignition-gazebo6::ignition-gazebo6 OpenSSL::Crypto nlohmann_json::nlohmann_json)
enable_testing()
add_test(NAME native-authorization COMMAND authorization-test)
add_test(NAME native-callback-target COMMAND native-test)
