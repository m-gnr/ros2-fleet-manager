cmake_minimum_required(VERSION 3.16)
project(c5_t3c1a LANGUAGES CXX)
find_package(ignition-gazebo6 6.18.0 EXACT REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(c5-authorized-zero-drive SHARED ../c5-b6b5/DiffDrive.cc)
target_compile_definitions(c5-authorized-zero-drive PRIVATE C5_SPLIT_ZERO_ONLY=1
 DiffDrive=C5SplitZeroDiffDrive DiffDrivePrivate=C5SplitZeroDiffDrivePrivate)
target_link_libraries(c5-authorized-zero-drive PRIVATE ignition-gazebo6::ignition-gazebo6)
add_executable(native-zero-worker worker.cc)
target_link_libraries(native-zero-worker PRIVATE ignition-gazebo6::ignition-gazebo6 OpenSSL::Crypto nlohmann_json::nlohmann_json)
add_executable(split-server server.cc)
target_link_libraries(split-server PRIVATE ignition-gazebo6::ignition-gazebo6)
add_executable(split-recorder record.cc)
target_link_libraries(split-recorder PRIVATE nlohmann_json::nlohmann_json)
add_library(c5-physics-observer SHARED physics.cc)
target_link_libraries(c5-physics-observer PRIVATE ignition-gazebo6::ignition-gazebo6)
add_executable(ipc-test test_ipc.cc)
target_link_libraries(ipc-test PRIVATE OpenSSL::Crypto nlohmann_json::nlohmann_json)
enable_testing()
add_test(NAME split-native-veto COMMAND ipc-test)
