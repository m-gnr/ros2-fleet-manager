Kapanış olayları ve SQLite son durumları ayrı toplandı.

```sh
for run in c23 auto full auto-repeat auto-final; do
  project="ros2-fleet-c334-task56-${run}"
  for service in simulation backend; do
    name="${project}-${service}-1"
    docker inspect --format '{{.Name}} exit={{.State.ExitCode}} oom={{.State.OOMKilled}} start={{.State.StartedAt}} end={{.State.FinishedAt}}' "$name"
    docker events --since "$(docker inspect --format '{{.State.StartedAt}}' "$name")" \
      --until "$(docker inspect --format '{{.State.FinishedAt}}' "$name")" \
      --filter "container=$name" --filter event=kill --filter event=die \
      --format '{{.TimeNano}} {{.Action}} signal={{index .Actor.Attributes "signal"}} exit={{index .Actor.Attributes "exitCode"}}'
  done
done

for run in auto full auto-repeat auto-final; do
  db=c3_3_3_t02-auto-routes.db
  if [ "$run" = full ]; then db=c3_3_3_t02.db; fi
  docker cp "ros2-fleet-c334-task56-${run}-backend-1:/tmp/$db" "/tmp/fleet-task56-${run}-sqlite.db"
done
```

DB kopyalarında aşağıdaki sorgular kullanıldı.

```sql
SELECT task_id, status, robot_id, assignment_epoch FROM tasks ORDER BY task_id;
SELECT task_id, json_extract(payload, '$.assigned_robot_id'), state FROM outbox_events ORDER BY task_id;
SELECT count(*) FROM processed_events WHERE outcome = 'ROUTE_APPLIED';
```
