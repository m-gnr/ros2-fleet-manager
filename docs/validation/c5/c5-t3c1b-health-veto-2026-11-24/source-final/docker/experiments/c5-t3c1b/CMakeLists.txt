cmake_minimum_required(VERSION 3.16)
project(c5_t3c1b LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
enable_testing()
add_subdirectory(../c5-t3c1a split)
foreach(target c5-authorized-zero-drive native-zero-worker split-recorder)
 target_compile_definitions(${target} PRIVATE C5_SPLIT_HEALTH_VETO=1)
endforeach()
foreach(target c5-authorized-zero-drive native-zero-worker split-recorder split-server c5-physics-observer ipc-test)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
add_executable(health-test test_health.cc)
target_compile_definitions(health-test PRIVATE C5_SPLIT_HEALTH_VETO=1)
target_link_libraries(health-test PRIVATE OpenSSL::Crypto nlohmann_json::nlohmann_json)
add_test(NAME health-receiver-and-final-veto COMMAND health-test)
