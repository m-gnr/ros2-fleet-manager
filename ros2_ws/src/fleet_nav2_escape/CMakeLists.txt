cmake_minimum_required(VERSION 3.8)
project(fleet_nav2_escape)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_behavior_tree REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)

add_library(fleet_escape_bt_nodes SHARED src/escape.cpp)
target_compile_definitions(fleet_escape_bt_nodes PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(fleet_escape_bt_nodes nav2_behavior_tree nav2_msgs nav_msgs
  rclcpp rclcpp_action tf2_geometry_msgs tf2_ros)
find_library(FOLLOW_PATH_LIBRARY nav2_follow_path_action_bt_node
  HINTS "${nav2_behavior_tree_DIR}/../../../lib" REQUIRED)
target_link_libraries(fleet_escape_bt_nodes ${FOLLOW_PATH_LIBRARY})
add_library(fleet_clearance_controller SHARED src/clearance_controller.cpp)
ament_target_dependencies(fleet_clearance_controller nav2_core nav2_msgs nav_msgs
  pluginlib rclcpp tf2_geometry_msgs tf2_ros)
pluginlib_export_plugin_description_file(nav2_core controllers.xml)
install(TARGETS fleet_escape_bt_nodes fleet_clearance_controller LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(clearance_test test/clearance_test.cpp)
  ament_target_dependencies(clearance_test nav2_msgs rclcpp tf2_geometry_msgs tf2_ros)
  add_test(NAME clearance_geometry COMMAND clearance_test)
endif()
ament_package()
