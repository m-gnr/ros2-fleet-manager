cmake_minimum_required(VERSION 3.8)
project(fleet_nav2_observer)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_behavior_tree REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(fleet_interfaces REQUIRED)
add_library(fleet_observed_follow_path_bt_node SHARED src/observed_follow_path.cpp)
target_compile_definitions(fleet_observed_follow_path_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(fleet_observed_follow_path_bt_node nav2_behavior_tree nav2_msgs rclcpp rclcpp_action fleet_interfaces)
find_library(FOLLOW_PATH_LIBRARY nav2_follow_path_action_bt_node
  HINTS "${nav2_behavior_tree_DIR}/../../../lib" REQUIRED)
target_link_libraries(fleet_observed_follow_path_bt_node ${FOLLOW_PATH_LIBRARY})
install(TARGETS fleet_observed_follow_path_bt_node LIBRARY DESTINATION lib)
ament_package()
