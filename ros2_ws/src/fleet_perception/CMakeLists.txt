cmake_minimum_required(VERSION 3.8)
project(fleet_perception)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(fleet_interfaces REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(observed_obstacle_layer SHARED src/observed_obstacle_layer.cpp src/peer_robot_layer.cpp)
ament_target_dependencies(observed_obstacle_layer nav2_costmap_2d pluginlib fleet_interfaces std_msgs tf2_msgs tf2_ros)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS observed_obstacle_layer LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_peer_robot_layer test/test_peer_robot_layer.cpp)
  ament_target_dependencies(test_peer_robot_layer nav2_costmap_2d pluginlib fleet_interfaces std_msgs tf2_msgs tf2_ros)
endif()
ament_package()
